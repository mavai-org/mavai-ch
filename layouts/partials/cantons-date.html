{{- /* Renders a date given as YYYY, YYYY-MM or YYYY-MM-DD, localised to the page language. */ -}} {{- $d := printf "%v" . -}} {{- if eq (len $d) 4 -}}{{ $d }} {{- else if eq (len $d) 7 -}}{{ time.Format "January 2006" (time.AsTime (printf "%s-01" $d)) }} {{- else -}}{{ time.Format ":date_long" (time.AsTime $d) }}{{- end -}}
